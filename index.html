// Обработка GET (чтение данных для дашборда) function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var productsSheet = ss.getSheetByName("СКЛАД") || ss.getSheetByName("Склад"); var shipmentsSheet = ss.getSheetByName("ВІДПРАВКИ") || ss.getSheetByName("Відправки"); var purchasesSheet = ss.getSheetByName("ЗАКУПІВЛІ") || ss.getSheetByName("Закупівлі"); var result = { status: "success", products: getProductsData(productsSheet), shipments: getShipmentsData(shipmentsSheet), purchases: getSheetDataUniversal(purchasesSheet) }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { // Никогда не даём скрипту вернуть необработанную HTML-страницу ошибки - // клиент всегда должен получать валидный JSON. return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString(), products: [], shipments: [], purchases: [] })).setMimeType(ContentService.MimeType.JSON); } } // Обработка POST (запись нового заказа из дашборда с точным распределением по колонкам) function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Порожній запит: відсутні дані (postData)" })).setMimeType(ContentService.MimeType.JSON); } var contents = e.postData.contents; var data = JSON.parse(contents); if (data.action === "addPurchase") { // Базова валідація вхідних даних if (!data.name || !data.id) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Не вказано ID або назву товару" })).setMimeType(ContentService.MimeType.JSON); } if (!data.qty || isNaN(data.qty) || Number(data.qty) <= 0) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Кількість має бути числом більше нуля" })).setMimeType(ContentService.MimeType.JSON); } var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("ЗАКУПІВЛІ") || ss.getSheetByName("Закупівлі"); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Лист 'ЗАКУПІВЛІ' не знайдено!" })).setMimeType(ContentService.MimeType.JSON); } // Формируем строку строго по порядку колонок от A до L (индексы от 0 до 11): // A (0): пусто // B (1): Назва товару (data.name) // C (2): ID CRM (data.id) // D (3): Замовлено, од (data.qty) // E (4): Номер ЕН (data.ttn) // F (5) - J (9): пусто // K (10): Постачальник (data.supplier) // L (11): Дата замовлення (data.date) var rowData = [ "", // A data.name || "", // B - Назва товару data.id || "", // C - ID CRM data.qty || 0, // D - Замовлено, од data.ttn || "", // E - Номер ЕН "", // F "", // G "", // H "", // I "", // J data.supplier || "",// K - Постачальник data.date || "" // L - Дата замовлення ]; sheet.appendRow(rowData); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } // Раніше тут функція нічого не повертала для невідомих action - // Apps Script віддавав порожню/некоректну відповідь. return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Невідома дія: " + data.action })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } // Чтение склада: передаем цену за единицу из колонки E function getProductsData(sheet) { if (!sheet) return []; var data = sheet.getDataRange().getValues(); if (data.length <= 2) return []; var rows = []; for (var i = 2; i < data.length; i++) { var foundId = data[i][1]; // Колонка B (индекс 1) - ID var foundName = data[i][2]; // Колонка C (индекс 2) - Назва товару var foundQty = data[i][3]; // Колонка D (индекс 3) - К-сть var foundBuyPrice = data[i][4]; // Колонка E (индекс 4) - Закуп, грн (цена за 1 шт) if (foundId !== "" && foundId !== null && foundId !== "ID" && foundId !== "ID CRM") { var qtyNum = (foundQty !== "" && !isNaN(foundQty)) ? Number(foundQty) : 0; var priceNum = (foundBuyPrice !== "" && !isNaN(foundBuyPrice)) ? Number(foundBuyPrice) : 0; rows.push({ id: foundId, name: foundName, qty: qtyNum, price: priceNum }); } } return rows; } // Гибкая функция для чтения отправлений (ВІДПРАВКИ) function getShipmentsData(sheet) { if (!sheet) return []; var data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; var headers = data[0]; var rows = []; for (var i = 1; i < data.length; i++) { var rawObj = {}; var hasValue = false; for (var j = 0; j < headers.length; j++) { var headerName = String(headers[j]).trim(); var cellValue = data[i][j]; if (headerName !== "") { rawObj[headerName] = cellValue; if (cellValue !== "" && cellValue !== null && cellValue !== undefined) { hasValue = true; } } } var foundId = ""; var foundName = ""; var foundQty = 0; var foundDate = ""; var foundTtn = ""; for (var key in rawObj) { var lowerKey = key.toLowerCase(); var val = rawObj[key]; if (lowerKey.includes('id') || lowerKey.includes('код') || lowerKey.includes('айди')) { if (val !== "" && val !== null) foundId = val; } if (lowerKey.includes('назв') || lowerKey.includes('товар')) { if (val !== "" && val !== null) foundName = val; } if (lowerKey.includes('замовл') || lowerKey.includes('кільк') || lowerKey.includes('оед') || lowerKey.includes('шт') || lowerKey.includes('qty')) { if (val !== "" && val !== null && !isNaN(val)) foundQty = Number(val); } if (lowerKey.includes('дат') || lowerKey.includes('date')) { if (val !== "" && val !== null) foundDate = val; } if (lowerKey.includes('ен') || lowerKey.includes('ттн') || lowerKey.includes('номер')) { if (val !== "" && val !== null) foundTtn = val; } } if (!foundId && data[i][2] !== undefined) foundId = data[i][2]; if (!foundName && data[i][1] !== undefined) foundName = data[i][1]; if (!foundQty && data[i][3] !== undefined && !isNaN(data[i][3])) foundQty = Number(data[i][3]); if (!foundDate && data[i][11] !== undefined) foundDate = data[i][11]; if (!foundTtn && data[i][4] !== undefined) foundTtn = data[i][4]; if (hasValue) { rows.push({ id: foundId, name: foundName, qty: foundQty, date: foundDate, ttn: foundTtn }); } } return rows; } // Универсальная функция для закупок function getSheetDataUniversal(sheet) { if (!sheet) return []; var data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; var headers = data[0]; var rows = []; for (var i = 1; i < data.length; i++) { var row = {}; var hasValue = false; for (var j = 0; j < headers.length; j++) { var headerName = String(headers[j]).trim(); var cellValue = data[i][j]; if (headerName !== "") { row[headerName] = cellValue; if (cellValue !== "" && cellValue !== null && cellValue !== undefined) { hasValue = true; } } } row.id = row['ID CRM'] || row['ID'] || row['айди'] || row['код'] || ''; row.name = row['НАЗВА ТОВАРУ'] || row['Назва'] || row['Название'] || ''; row.qty = row['ЗАМОВЛЕНО, ОД'] !== undefined ? row['ЗАМОВЛЕНО, ОД'] : (row['Кількість'] !== undefined ? row['Кількість'] : 0); row.date = row['Дата замовлення'] || row['Дата'] || ''; row.ttn = row['НОМЕР ЕН'] || row['ТТН'] || row['ЕН'] || ''; row.supplier = row['Постачальник'] || ''; if (hasValue) { rows.push(row); } } return rows; }
