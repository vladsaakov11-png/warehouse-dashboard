// Функция для корректного чтения даты из Google Таблиц function parseSheetDate(dateInput) { if (!dateInput) return null; let d = new Date(dateInput); if (!isNaN(d.getTime())) return d.getTime(); // Парсим формат DD.MM.YYYY HH:mm:ss или DD.MM.YYYY const str = String(dateInput).trim(); const parts = str.split(/[\sT]+/); const dateParts = parts[0].split('.'); if (dateParts.length === 3) { const day = parseInt(dateParts[0], 10); const month = parseInt(dateParts[1], 10) - 1; const year = parseInt(dateParts[2], 10); let timeParts = [0, 0, 0]; if (parts[1]) { timeParts = parts[1].split(':').map(n => parseInt(n, 10) || 0); } return new Date(year, month, day, timeParts[0], timeParts[1], timeParts[2] || 0).getTime(); } return null; } // Обновленный подсчет отправок function getShipmentCountForProduct(productId) { if (!allShipments || !Array.isArray(allShipments)) return 0; const now = new Date().getTime(); let hoursLimit = 24; if (currentPeriod === '7d') hoursLimit = 24 * 7; if (currentPeriod === '30d') hoursLimit = 24 * 30; const timeThreshold = now - (hoursLimit * 60 * 60 * 1000); const targetId = String(productId).replace('#', '').trim(); return allShipments.reduce((sum, s) => { const sId = String(s.id).replace('#', '').trim(); if (sId === targetId) { const shipTime = parseSheetDate(s.date); if (shipTime && shipTime >= timeThreshold) { return sum + (parseInt(s.qty, 10) || 1); } } return sum; }, 0); }
